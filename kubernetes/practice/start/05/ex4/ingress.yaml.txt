# 舊專案的 Ingress（給 Ingress NGINX 用）。只讀不套用：副檔名是 .txt，kubectl apply 不會碰它
# 兩個主機：api.cafe.test（orders 服務）、www.cafe.test（網站與菜單）
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: cafe
spec:
  ingressClassName: nginx
  rules:
  - host: api.cafe.test
    http:
      paths:
      - path: /orders
        pathType: Prefix
        backend:
          service:
            name: orders
            port:
              number: 80
      - path: /healthz
        pathType: Exact
        backend:
          service:
            name: orders
            port:
              number: 80
  - host: www.cafe.test
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: web
            port:
              number: 80
---
# 菜單服務自己的路由是 /items、/specials，對外卻要掛在 /menu 底下，
# 所以用 Ingress NGINX 的 rewrite-target 把 /menu 拿掉再轉送：/menu/items → /items、/menu → /
# annotation 對整個 Ingress 物件生效，所以要 rewrite 的路徑單獨放一個 Ingress
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: cafe-menu
  annotations:
    nginx.ingress.kubernetes.io/use-regex: "true"
    nginx.ingress.kubernetes.io/rewrite-target: /$2
spec:
  ingressClassName: nginx
  rules:
  - host: www.cafe.test
    http:
      paths:
      - path: /menu(/|$)(.*)
        pathType: ImplementationSpecific
        backend:
          service:
            name: menu
            port:
              number: 80
